<script>
  document.addEventListener("DOMContentLoaded", function() {
    // Animate research stats
    function animateStats() {
      const statNumbers = document.querySelectorAll('[data-count]');
      statNumbers.forEach(stat => {
        const target = parseInt(stat.dataset.count) || 7; // Default fallback
        const increment = target / 60;
        let current = 0;
        
        const timer = setInterval(() => {
          current += increment;
          if (current >= target) {
            current = target;
            clearInterval(timer);
          }
          stat.textContent = Math.floor(current);
        }, 50);
      });
    }
    
    // Initialize stats animation after a delay
    setTimeout(animateStats, 500);
    
    // Domain filtering
    const domainTags = document.querySelectorAll('.domain-tag');
    const projectCards = document.querySelectorAll('.project-card');
    
    domainTags.forEach(tag => {
      tag.addEventListener('click', function() {
        // Update active state
        domainTags.forEach(t => t.classList.remove('active'));
        this.classList.add('active');
        
        const domain = this.dataset.domain;
        
        // Filter projects (basic implementation - can be enhanced)
        projectCards.forEach(card => {
          if (domain === 'all') {
            card.style.display = 'flex';
          } else {
            // Simple filtering based on project content
            const cardText = card.textContent.toLowerCase();
            if (domain === 'medical-robotics' && (cardText.includes('medical') || cardText.includes('surgical') || cardText.includes('robotic'))) {
              card.style.display = 'flex';
            } else if (domain === 'soft-robotics' && cardText.includes('soft')) {
              card.style.display = 'flex';
            } else if (domain === 'simulation' && cardText.includes('simulation')) {
              card.style.display = 'flex';
            } else if (domain === 'ai-healthcare' && (cardText.includes('ai') || cardText.includes('digital'))) {
              card.style.display = 'flex';
            } else {
              card.style.display = 'none';
            }
          }
        });
      });
    });
    
    console.log('🚀 Enhanced research page loaded successfully!');
  });
</script>
